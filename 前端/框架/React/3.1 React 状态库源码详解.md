### createStore

创建容器。

```JavaScript
function createStore(reducer, preloadedState, enhancer) {
  /**
   * 预处理
   * 1. preloadedState 为函数且 enhancer 不存在，说明 enhancer 为 preloadedState
   * 2. 存在 enhancer，说明需要对 createStore 进行 enhancer 增强
   */
  if (typeof preloadedState === 'function' && typeof enhancer === 'undefined') {
    enhancer = preloadedState
    preloadedState = undefined
  }
  if (typeof enhancer !== 'undefined') {
    return enhancer(createStore)(reducer, preloadedState)
  }

  let currentReducer = reducer         // reducer 的快照
  let currentState = preloadedState    // 仓库状态
  let currentListeners = new Map()     // 订阅映射
  let nextListeners = currentListeners // 临时订阅映射，dispatch 时使用
  let listenerIdCounter = 0            // 订阅者数量
  let isDispatching = false            // 是否在派发

  function ensureCanMutateNextListeners() {
    if (nextListeners === currentListeners) {
      nextListeners = new Map()
      currentListeners.forEach((listener, key) => {
        nextListeners.set(key, listener)
      })
    }
  }

  /**
   * 获取仓库状态
   */
  function getState() {
    return currentState
  }

  /**
   * 订阅
   */
  function subscribe(listener) {
    let isSubscribed = true
    ensureCanMutateNextListeners()
    const listenerId = listenerIdCounter++
    nextListeners.set(listenerId, listener)
    return function unsubscribe() {
      if (!isSubscribed) {
        return
      }
      isSubscribed = false
      ensureCanMutateNextListeners()
      nextListeners.delete(listenerId)
      currentListeners = null
    }
  }

  /**
   * 派发action，修改仓库状态
   */
  function dispatch(action) {
    try {
      isDispatching = true
      currentState = currentReducer(currentState, action)
    } finally {
      isDispatching = false
    }
    const listeners = (currentListeners = nextListeners)
    listeners.forEach(listener => {
      listener()
    })
    return action
  }

  /**
   * 替换reducer
   */
  function replaceReducer(nextReducer) {
    currentReducer = nextReducer
    dispatch({ type: ActionTypes.REPLACE })
  }

  dispatch({ type: ActionTypes.INIT })

  const store = {
    dispatch: dispatch,
    subscribe,
    getState,
    replaceReducer,
    [$$observable]: observable
  }
  return store
}
```

### combineReducers

合并 reducer。

```JavaScript
/**
 * 合并 reducer 函数
 */
function combineReducers(reducers) {
  /**
   * 预处理，过滤不合法的键值对（值不为 reducer 函数）
   */
  const reducerKeys = Object.keys(reducers) // reducer 函数的 key 集合
  const finalReducers = {} // 有效的 reducer 函数集合
  for (let i = 0; i < reducerKeys.length; i++) {
    const key = reducerKeys[i]
    if (typeof reducers[key] === 'function') {
      finalReducers[key] = reducers[key]
    }
  }
  const finalReducerKeys = Object.keys(finalReducers) // 有效的 reducer 函数的 key 集合

  /**
   * 合并后的 combination 函数，调用时遍历所有子 reducer 生成新 state
   */
  return function combination(state = {}, action) {
    let hasChanged = false // 标识 state 是否改变
    const nextState = {}
    for (let i = 0; i < finalReducerKeys.length; i++) {
      const key = finalReducerKeys[i]                              // 当前遍历的 reducer 的 key
      const reducer = finalReducers[key]                           // 当前遍历的 reducer
      const previousStateForKey = state[key]                       // 当前 reducer 对应的 state 状态
      const nextStateForKey = reducer(previousStateForKey, action) // 执行新的 reducer 获取新值
      nextState[key] = nextStateForKey
      hasChanged = hasChanged || nextStateForKey !== previousStateForKey
    }
    hasChanged = hasChanged || finalReducerKeys.length !== Object.keys(state).length
    return hasChanged ? nextState : state
  }
}
```

### bindActionCreators

```JavaScript
/**
 * 包裹单个 action creator 函数
 */
function bindActionCreator(actionCreator, dispatch) {
  return function (this, ...args) {
    return dispatch(actionCreator.apply(this, args))
  }
}

/**
 * 包裹 action creator 函数集合
 */
function bindActionCreators(actionCreators, dispatch) {
  // 传入的是单个函数，直接返回包裹后的函数
  if (typeof actionCreators === 'function') {
    return bindActionCreator(actionCreators, dispatch)
  }

  // 传入的是对象，返回包裹后的对象
  const boundActionCreators = {}
  for (const key in actionCreators) {
    const actionCreator = actionCreators[key]
    if (typeof actionCreator === 'function') {
      boundActionCreators[key] = bindActionCreator(actionCreator, dispatch)
    }
  }
  return boundActionCreators
}
```

## 中间件

### applyMiddleware

应用中间件，返回增强后的 createStore。

```JavaScript
/**
 * 组合多个函数，从右往左依次执行
 * @param  {...Function} funcs 函数集合
 * @returns {Function} 组合后的函数
 */
function compose(...funcs) {
  /**
   * 默认返回的函数
   */
  if (funcs.length === 0) {
    return (arg) => arg
  }

  if (funcs.length === 1) {
    return funcs[0]
  }

  return funcs.reduce((a, b) => (...args) => a(b(...args)))
}

/**
 * 应用中间件
 * @param  {...any} middlewares 中间件集合
 * @returns {StoreEnhancer < any >} 增强后的 createStore 函数
 */
function applyMiddleware(...middlewares) {
  return createStore => (reducer, preloadedState) => {
    const store = createStore(reducer, preloadedState)
    let dispatch = () => {
      throw new Error('Dispatching while constructing your middleware is not allowed. ' + 'Other middleware would not be applied to this dispatch.')
    }

    const middlewareAPI = {
      getState: store.getState,
      dispatch: (action, ...args) => dispatch(action, ...args)
    }
    const chain = middlewares.map(middleware => middleware(middlewareAPI))
    dispatch = compose(...chain)(store.dispatch)

    return {
      ...store,
      dispatch
    }
  }
}
```

### configureStore

对 createStore 的高级封装，集成了默认 reducer 合并、中间件、DevTools 支持、默认增强器等功能，目的是简化 Redux 的配置流程。

```JavaScript
/**
 * 构建默认中间件构造函数
 * @description 根据用户传入的配置项，动态构建并返回一组默认中间件（如 thunk、immutable 检查、序列化检查等），主要用于开发环境调试和最佳实践内建支持
 */
const buildGetDefaultMiddleware = () =>
  function getDefaultMiddleware(options) {
    /**
     * 默认中间件配置
     * - true：开启
     * - false：关闭
     * - object：开启并使用自定义配置
     */
    const {
      thunk = true,
      immutableCheck = true,
      serializableCheck = true,
      actionCreatorCheck = true,
    } = options ?? {}

    /**
     * 添加默认中间件
     */
    let middlewareArray = new Tuple() // 中间件数组

    // redux-thunk
    if (thunk) {
      if (isBoolean(thunk)) {
        middlewareArray.push(thunkMiddleware)
      } else {
        middlewareArray.push(withExtraArgument(thunk.extraArgument))
      }
    }

    // 开发环境下开启调试性中间件
    if (process.env.NODE_ENV !== 'production') {
      // redux-immutable-state-invariant：检查 reducer 是否非法修改了 state
      if (immutableCheck) {
        let immutableOptions = {}
        if (!isBoolean(immutableCheck)) {
          immutableOptions = immutableCheck
        }
        middlewareArray.unshift(
          createImmutableStateInvariantMiddleware(immutableOptions),
        )
      }

      // redux-serializable-state-invariant：检查 action/state 是否是可序列化的
      if (serializableCheck) {
        let serializableOptions = {}
        if (!isBoolean(serializableCheck)) {
          serializableOptions = serializableCheck
        }
        middlewareArray.push(
          createSerializableStateInvariantMiddleware(serializableOptions),
        )
      }

      // redux-action-creator-check：检查 action 创建器是否合法
      if (actionCreatorCheck) {
        let actionCreatorOptions = {}
        if (!isBoolean(actionCreatorCheck)) {
          actionCreatorOptions = actionCreatorCheck
        }
        middlewareArray.unshift(
          createActionCreatorInvariantMiddleware(actionCreatorOptions),
        )
      }
    }

    return middlewareArray
  }

/**
 * 创建 store
 * @description 对 createStore 的高级封装，集成了默认 reducer 合并、中间件、DevTools 支持、默认增强器等功能，目的是简化 Redux 的配置流程
 */
function configureStore(options) {
  /**
   * 获取默认中间件构造函数
   * @description 内部封装了redux-thunk、serializableCheck（检查 action/state 是否可序列化）、immutableCheck（检查 state 是否被非法 mutate）
   */
  const getDefaultMiddleware = buildGetDefaultMiddleware()

  const {
    reducer = undefined,        // 调度器
    middleware,                 // 中间件集合
    devTools = true,            // 是否开启 Redux DevTools
    preloadedState = undefined, // 初始化状态
    enhancers = undefined,      // 附加 enhancer，类似 applyMiddleware + otherCompose
  } = options || {}

  /**
   * 处理 reducer，函数直接返回，对象则合并
   */
  let rootReducer
  if (typeof reducer === 'function') {
    rootReducer = reducer
  } else if (isPlainObject(reducer)) {
    rootReducer = combineReducers(reducer)
  }

  /**
   * 处理中间件，函数直接返回，数组则合并
   */
  let finalMiddleware
  if (typeof middleware === 'function') {
    finalMiddleware = middleware(getDefaultMiddleware)
  } else {
    finalMiddleware = getDefaultMiddleware()
  }

  /**
   * 自定义 enhancer
   */
  let finalCompose = compose
  const middlewareEnhancer = applyMiddleware(...finalMiddleware)
  const getDefaultEnhancers = buildGetDefaultEnhancers(middlewareEnhancer)
  let storeEnhancers =
    typeof enhancers === 'function'
      ? enhancers(getDefaultEnhancers)
      : getDefaultEnhancers()
  const composedEnhancer = finalCompose(...storeEnhancers)

  return createStore(rootReducer, preloadedState, composedEnhancer)
}
```

### createAction

创建 action 构建器。

```TypeScript
/**
 * 创建 action 构建器
 * @param {*} type action 类型
 * @param {*} prepareAction 预处理函数
 * @returns actionCreator，用于构建 action 对象
 */
function createAction(type, prepareAction) {
  function actionCreator(...args) {
    if (prepareAction) {
      let prepared = prepareAction(...args)
      return {
        type,
        payload: prepared.payload,
        ...('meta' in prepared && { meta: prepared.meta }),
        ...('error' in prepared && { error: prepared.error }),
      }
    }
    return { type, payload: args[0] }
  }
  actionCreator.toString = () => `${type}`
  actionCreator.type = type
  actionCreator.match = (action) => isAction(action) && action.type === type
  return actionCreator
}
```

### createReducer

创建 reducer，封装了 reducer 声明、matcher 匹配、Immer 代理等高级逻辑，目的是让你写的 reducer 又短又强大。

```JavaScript
/**
 * 创建 reducer
 * @description 构建一个支持 可变写法（通过 Immer）、支持 matcher、defaultCase 的 reducer 函数，是 createSlice 背后的执行核心
 * @param {*} initialState 初始状态
 * @param {*} mapOrBuilderCallback 构建 reducer 函数的回调函数
 */
function createReducer(initialState, mapOrBuilderCallback) {
  /**
   * 处理 reducers 声明结构
   * 将 builder 写法（或普通对象形式）统一转换为内部标准格式：
   * - actionsMap：{ [type]: caseReducer }
   * - finalActionMatchers：[{ matcher, reducer }]
   * - finalDefaultCaseReducer：默认 caseReducer（addDefaultCase）
   */
  let [actionsMap, finalActionMatchers, finalDefaultCaseReducer] = executeReducerBuilderCallback(mapOrBuilderCallback)

  /**
   * 封装 getInitialState
   */
  let getInitialState
  if (isStateFunction(initialState)) {
    getInitialState = () => freezeDraftable(initialState())
  } else {
    const frozenInitialState = freezeDraftable(initialState)
    getInitialState = () => frozenInitialState
  }

  /**
   * 构造最终 reducer 函数
   */
  function reducer(state = getInitialState(), action) {
    // 收集所有命中的 reducer
    let caseReducers = [
      actionsMap[action.type],
      ...finalActionMatchers
        .filter(({ matcher }) => matcher(action))
        .map(({ reducer }) => reducer),
    ]
    if (caseReducers.filter((cr) => !!cr).length === 0) {
      caseReducers = [finalDefaultCaseReducer]
    }

    /**
     * 按顺序执行 caseReducer
     * 1. 当前 state 是 Immer 的 draft：直接调用 reducer
     * 2. state 是不可 draft 类型（如数字、null）：直接调用 reducer
     * 3. 普通对象或数组：使用 Immer 的 createNextState(state, draft => reducer(draft))
     */
    return caseReducers.reduce((previousState, caseReducer) => {
      if (caseReducer) {
        if (isDraft(previousState)) {
          const draft = previousState
          const result = caseReducer(draft, action)
          if (result === undefined) {
            return previousState
          }
          return result
        } else if (!isDraftable(previousState)) {
          const result = caseReducer(previousState, action)
          if (result === undefined) {
            if (previousState === null) {
              return previousState
            }
          }
          return result
        } else {
          return createNextState(previousState, (draft) => {
            return caseReducer(draft, action)
          })
        }
      }

      return previousState
    }, state)
  }

  reducer.getInitialState = getInitialState

  return reducer
}
```

### createSlice

创建切片。

> 切片可以理解为 State \+ Action \+ Reducer，但不等于 Store。

```JavaScript
/**
 * 构建定制版的 createSlice
 * @description 用于扩展 Redux Toolkit 的切片创建行为
 */
function buildCreateSlice({ creators } = {}) {
  const cAT = creators?.asyncThunk?.[asyncThunkSymbol]
  return function createSlice(options) {
    const { name, reducerPath = name } = options

    /**
     * 构建 Redux 切片的 reducer 集合
     * @description 用于创建 Redux 切片的 reducer 集合，支持函数形式和对象形式
     * - 函数形式：接受一个参数，用于构建 reducer 集合
     * - 对象形式：直接传入 reducer 集合
     */
    const reducers = (typeof options.reducers === 'function' ? options.reducers(buildReducerCreators()) : options.reducers) || {}
    const reducerNames = Object.keys(reducers) // reducer 的 key 集合

    /**
     * 构建 Redux 切片的 action 创建函数集合
     */
    const context = {
      sliceCaseReducersByName: {}, // 所有根据 action type 注册的 reducer
      sliceCaseReducersByType: {}, // 自动生成的 action 创建函数
      actionCreators: {},          // 支持 matcher 规则
      sliceMatchers: [],           // 用于导出的 caseReducers（方便调试或测试）
    }
    const contextMethods = {
      addCase(typeOrActionCreator, reducer) {
        const type = typeof typeOrActionCreator === 'string' ? typeOrActionCreator : typeOrActionCreator.type
        context.sliceCaseReducersByType[type] = reducer
        return contextMethods
      },
      addMatcher(matcher, reducer) {
        context.sliceMatchers.push({ matcher, reducer })
        return contextMethods
      },
      exposeAction(name, actionCreator) {
        context.actionCreators[name] = actionCreator
        return contextMethods
      },
      exposeCaseReducer(name, reducer) {
        context.sliceCaseReducersByName[name] = reducer
        return contextMethods
      },
    }

    /**
     * 遍历 reducers，处理普通和异步 thunk
     * - 处理普通 reducer：根据 reducer 定义创建 action 创建函数，并将其添加到 actionCreators 中
     * - 处理异步 thunk：根据 thunk 定义创建 action 创建函数，并将其添加到 actionCreators 中
     */
    reducerNames.forEach((reducerName) => {
      const reducerDefinition = reducers[reducerName] // 当前遍历的 reducer
      const reducerDetails = {
        reducerName,
        type: getType(name, reducerName),
        createNotation: typeof options.reducers === 'function',
      }
      if (isAsyncThunkSliceReducerDefinition(reducerDefinition)) {
        handleThunkCaseReducerDefinition(
          reducerDetails,
          reducerDefinition,
          contextMethods,
          cAT,
        )
      } else {
        handleNormalReducerDefinition(
          reducerDetails,
          reducerDefinition,
          contextMethods,
        )
      }
    })

    /**
     * 构建最终的 reducer 函数
     */
    function buildReducer() {
      const [
        extraReducers = {},
        actionMatchers = [],
        defaultCaseReducer = undefined,
      ] = typeof options.extraReducers === 'function' ? executeReducerBuilderCallback(options.extraReducers) : [options.extraReducers]

      const finalCaseReducers = {
        ...extraReducers,
        ...context.sliceCaseReducersByType,
      }

      return createReducer(options.initialState, (builder) => {
        for (let key in finalCaseReducers) {
          builder.addCase(key, finalCaseReducers[key])
        }
        for (let sM of context.sliceMatchers) {
          builder.addMatcher(sM.matcher, sM.reducer)
        }
        for (let m of actionMatchers) {
          builder.addMatcher(m.matcher, m.reducer)
        }
        if (defaultCaseReducer) {
          builder.addDefaultCase(defaultCaseReducer)
        }
      })
    }
    const selectSelf = (state) => state
    const injectedSelectorCache = new Map()
    const injectedStateCache = new WeakMap()
    let _reducer
    function reducer(state, action) {
      if (!_reducer) _reducer = buildReducer()
      return _reducer(state, action)
    }
    function getInitialState() {
      if (!_reducer) _reducer = buildReducer()
      return _reducer.getInitialState()
    }

    /**
     * 动态 selector 构造
     * @param {*} reducerPath 切片路径
     * @param {*} injected 是否注入
     * @returns
     */
    function makeSelectorProps(
      reducerPath,
      injected = false,
    ) {
      function selectSlice(state) {
        let sliceState = state[reducerPath]
        if (typeof sliceState === 'undefined') {
          if (injected) {
            sliceState = getOrInsertComputed(
              injectedStateCache,
              selectSlice,
              getInitialState,
            )
          }
        }
        return sliceState
      }

      function getSelectors(selectState = selectSelf) {
        const selectorCache = getOrInsertComputed(
          injectedSelectorCache,
          injected,
          () => new WeakMap(),
        )

        return getOrInsertComputed(selectorCache, selectState, () => {
          const map = {}
          for (const [name, selector] of Object.entries(
            options.selectors ?? {},
          )) {
            map[name] = wrapSelector(
              selector,
              selectState,
              () =>
                getOrInsertComputed(
                  injectedStateCache,
                  selectState,
                  getInitialState,
                ),
              injected,
            )
          }
          return map
        })
      }
      return {
        reducerPath,
        getSelectors,
        get selectors() {
          return getSelectors(selectSlice)
        },
        selectSlice,
      }
    }

    const slice = {
      name,
      reducer,
      actions: context.actionCreators,
      caseReducers: context.sliceCaseReducersByName,
      getInitialState,
      ...makeSelectorProps(reducerPath),
      // slice 注入机制（用于懒加载）
      injectInto(injectable, { reducerPath: pathOpt, ...config } = {}) {
        const newReducerPath = pathOpt ?? reducerPath
        injectable.inject({ reducerPath: newReducerPath, reducer }, config)
        return {
          ...slice,
          ...makeSelectorProps(newReducerPath, true),
        }
      },
    }
    return slice
  }
}
```

## Thunk

### createAsyncThunk

创建带生命周期 action 的异步 thunk。

```JavaScript
/**
 * 创建带生命周期 action 的异步 thunk
 * @description 大幅简化了 Redux 中异步操作（如发请求、状态切换、错误处理）的流程，并自动生成如下三个阶段的 action：pending、fulfilled、rejected
 * @returns createAsyncThunk，异步 action creator，自动派发生命周期 actions，支持取消、中断、错误处理、条件控制、附加元数据等高级功能
 */
const createAsyncThunk = (() => {
  function createAsyncThunk(typePrefix, payloadCreator, options) {
    /**
     * 创建 3 个生命周期 action（标准 Redux action）
     * 每个 action 都会带 meta 字段，标明：
     * - 当前参数 arg
     * - 当前请求 ID（用于中断或去重）
     * - 请求状态：pending / fulfilled / rejected
     */
    const fulfilled = createAction(
      typePrefix + '/fulfilled',
      (payload, requestId, arg, meta) => ({
        payload,
        meta: {
          ...((meta) || {}),
          arg,
          requestId,
          requestStatus: 'fulfilled',
        },
      }),
    )
    const pending = createAction(
      typePrefix + '/pending',
      (requestId, arg, meta) => ({
        payload: undefined,
        meta: {
          ...((meta) || {}),
          arg,
          requestId,
          requestStatus: 'pending',
        },
      }),
    )
    const rejected = createAction(
      typePrefix + '/rejected',
      (error, requestId, arg, payload, meta) => ({
        payload,
        error: ((options && options.serializeError) || miniSerializeError)(
          error || 'Rejected',
        ),
        meta: {
          ...((meta) || {}),
          arg,
          requestId,
          rejectedWithValue: !!payload,
          requestStatus: 'rejected',
          aborted: error?.name === 'AbortError',
          condition: error?.name === 'ConditionError',
        },
      }),
    )

    /**
     * 异步 thunk action creator
     * - 支持中间件调用
     * - 接收 dispatch, getState, extra
     * - 返回一个 Promise（可 await）
     */
    function actionCreator(arg, { signal } = {}) {
      return (dispatch, getState, extra) => {
        const requestId = options?.idGenerator ? options.idGenerator(arg) : nanoid()
        const abortController = new AbortController()
        let abortHandler
        let abortReason
        function abort(reason) {
          abortReason = reason
          abortController.abort()
        }

        if (signal) {
          if (signal.aborted) {
            abort(externalAbortMessage)
          } else {
            signal.addEventListener(
              'abort',
              () => abort(externalAbortMessage),
              { once: true },
            )
          }
        }

        /**
         * 异步执行主流程
         * @description 自动封装 try/catch/finally 逻辑，派发状态 action
         */
        const promise = (async function () {
          let finalAction
          try {
            let conditionResult = options?.condition?.(arg, { getState, extra })
            if (isThenable(conditionResult)) {
              conditionResult = await conditionResult
            }
            const abortedPromise = new Promise((_, reject) => {
              abortHandler = () => {
                reject({
                  name: 'AbortError',
                  message: abortReason || 'Aborted',
                })
              }
              abortController.signal.addEventListener('abort', abortHandler)
            })
            dispatch(
              pending(
                requestId,
                arg,
                options?.getPendingMeta?.(
                  { requestId, arg },
                  { getState, extra },
                ),
              ),
            )
            finalAction = await Promise.race([
              abortedPromise,
              Promise.resolve(
                payloadCreator(arg, {
                  dispatch,
                  getState,
                  extra,
                  requestId,
                  signal: abortController.signal,
                  abort,
                  rejectWithValue: ((
                    value,
                    meta,
                  ) => {
                    return new RejectWithValue(value, meta)
                  }),
                  fulfillWithValue: ((value, meta) => {
                    return new FulfillWithMeta(value, meta)
                  }),
                }),
              ).then((result) => {
                if (result instanceof RejectWithValue) {
                  throw result
                }
                if (result instanceof FulfillWithMeta) {
                  return fulfilled(result.payload, requestId, arg, result.meta)
                }
                return fulfilled(result, requestId, arg)
              }),
            ])
          } catch (err) {
            finalAction =
              err instanceof RejectWithValue
                ? rejected(null, requestId, arg, err.payload, err.meta)
                : rejected(err, requestId, arg)
          } finally {
            if (abortHandler) {
              abortController.signal.removeEventListener('abort', abortHandler)
            }
          }
          const skipDispatch = options && !options.dispatchConditionRejection && rejected.match(finalAction) && (finalAction).meta.condition
          if (!skipDispatch) {
            dispatch(finalAction)
          }
          return finalAction
        })()
        return Object.assign(promise, {
          abort,
          requestId,
          arg,
          unwrap() {
            return promise.then < any > (unwrapResult)
          },
        })
      }
    }

    return Object.assign(
      actionCreator,
      {
        pending,
        rejected,
        fulfilled,
        settled: isAnyOf(rejected, fulfilled),
        typePrefix,
      },
    )
  }
  createAsyncThunk.withTypes = () => createAsyncThunk

  return createAsyncThunk
})()
```

```JavaScript
const createStoreImpl = (createState) => {
  let state // store 的状态
  const listeners = new Set() // 订阅函数集合

  /**
   * 修改状态
   * @param {any} partial 新状态
   * @param {any} replace 是否替换状态，false：合并；true：替换
   */
  const setState = (partial, replace) => {
    // 执行传入的 partial 函数，生成新的状态
    const nextState = typeof partial === 'function'
      ? partial(state)
      : partial

    if (!Object.is(nextState, state)) {
      const previousState = state
      // 替换或合并状态
      state = (replace ?? (typeof nextState !== 'object' || nextState === null))
        ? nextState
        : Object.assign({}, state, nextState)
      // 遍历执行订阅函数
      listeners.forEach((listener) => listener(state, previousState))
    }
  }

  /**
   * 获取当前状态
   */
  const getState = () => state

  /**
   * 获取初始化状态
   */
  const getInitialState = () => initialState

  /**
   * 订阅函数
   * @param {any} listener 订阅函数
   * @returns 销毁函数
   */
  const subscribe = (listener) => {
    listeners.add(listener)
    return () => listeners.delete(listener)
  }

  const api = { setState, getState, getInitialState, subscribe }

  // 执行传入的 createState 函数，生成初始化状态
  const initialState = (state = createState(setState, getState, api))

  return api
}

export const createStore = (createState) => createState ? createStoreImpl(createState) : createStoreImpl
```

### useAtomValue

实现 atom 值的订阅机制。

```TypeScript
function useAtomValue(atom, options) {
  /**
   * 支持多种异步处理模式
   * 通过 options.unstable_promiseStatus 控制 Promise 处理行为：
   * - 开启时：附加状态元数据（pending/fulfilled/rejected）
   * - 关闭时：标准 Suspense 行为
   */
  const { delay, unstable_promiseStatus: promiseStatus = !React.use } =
    options || {}
  const store = useStore(options)

  /**
   * 状态初始化与更新
   * - 使用 useReducer 存储当前值 + store + atom 的三元组
   * - 通过 Object.is 严格值比较判断是否需要更新
   */
  const [[valueFromReducer, storeFromReducer, atomFromReducer], rerender] =
    useReducer(
      (prev) => {
        const nextValue = store.get(atom)
        if (
          Object.is(prev[0], nextValue) &&
          prev[1] === store &&
          prev[2] === atom
        ) {
          return prev
        }
        return [nextValue, store, atom]
      },
      undefined,
      () => [store.get(atom), store, atom],
    )

  let value = valueFromReducer
  if (storeFromReducer !== store || atomFromReducer !== atom) {
    // 引用一致性检查（确保 store 或 atom 引用变更时立即更新）
    rerender()
    value = store.get(atom)
  }

  /**
   * 订阅状态变化
   * - 使用 store 的 sub 方法订阅 atom 变化
   * - 处理多种更新情况：
   *   - Promise 状态跟踪（当 promiseStatus: true）
   *   - 延迟渲染（delay 选项处理批更新）
   *   - 即时更新
   */
  useEffect(() => {
    const unsub = store.sub(atom, () => {
      if (promiseStatus) {
        const value = store.get(atom)
        if (isPromiseLike(value)) {
          attachPromiseStatus(
            createContinuablePromise(value, () => store.get(atom)),
          )
        }
      }
      if (typeof delay === 'number') {
        // 更新批处理与延迟渲染（优化高频更新场景，如：动画）
        setTimeout(rerender, delay)
        return
      }
      rerender()
    })
    rerender()
    return unsub
  }, [store, atom, delay, promiseStatus])

  /**
   * 调试支持
   * - 在 React DevTools 中显示当前 atom 值
   */
  useDebugValue(value)
  
  /**
   * 异步值处理（Suspense 集成）
   * - 检测 Promise 类值（Async/Promise atoms）
   * - 创建"可续传 Promise"确保最新状态
   * - 使用 React 的 use Hook（Experimental API）触发 Suspense
   */
  if (isPromiseLike(value)) {
    /**
     * 可续传 Promise 机制
     * - 包装原子值 Promise，确保总是获取最新值
     * - 支持在 Promise 解析前值发生变更的场景
     */
    const promise = createContinuablePromise(value, () => store.get(atom))
    if (promiseStatus) {
      attachPromiseStatus(promise)
    }
    return use(promise)
  }
  return value
}
```

#### [为什么不使用 useSyncExternalStore？](https://github.com/pmndrs/jotai/discussions/2137)

useSyncExternalStore 能够解决 [tearing](https://github.com/reactwg/react-18/discussions/69) 问题，但在处理 Promise 时表现不佳，并且 useSyncExternalStore 会导致 React 的时间切片失效。

#### [为什么使用 useReducer 而不是 useState？](https://github.com/pmndrs/jotai/discussions/2137#discussioncomment-9826358)

依赖 useReducer 的懒更新，即仅在组件首次渲染时执行一次初始状态计算。

```JavaScript
/**
 * 读取原子值

const returnAtomValue = (atomState) => {
  return atomState.v

const buildStore = (
  /** 原子状态存储 */
  atomStateMap = new WeakMap(),
  /** 原子挂载状态（组件依赖关系） */
  mountedMap = new WeakMap(),
  /** 本次更新影响的原子集合 */
  changedAtoms = new Set(),
) => {
  /**
   * 挂载原子依赖
   */
  const mountDependencies = (atom) => {
    const atomState = ensureAtomState(atom)
    const mounted = mountedMap.get(atom)
    // 已经挂载，遍历更新依赖
    if (mounted) {
      for (const [a, n] of atomState.d) {
        if (!mounted.d.has(a)) {
          const aState = ensureAtomState(a)
          const aMounted = mountAtom(a)
          aMounted.t.add(atom)
          mounted.d.add(a)
          if (n !== aState.n) {
            changedAtoms.add(a)
          }
        }
      }
      // 没有挂载，说明原子已经销毁，遍历删除依赖
      for (const a of mounted.d || []) {
        if (!atomState.d.has(a)) {
          mounted.d.delete(a)
          const aMounted = unmountAtom(a)
          aMounted?.t.delete(atom)
        }
      }
    }
  }

  /**
   * 获取原子状态，如果没有，则创建一个未初始化原子状态（没有 v 和 e）
   */
  const ensureAtomState = ((atom) => {
    let atomState = atomStateMap.get(atom)
    if (!atomState) {
      atomState = { d: new Map(), p: new Set(), n: 0 }
      atomStateMap.set(atom, atomState)
    }
    return atomState
  })

  /**
   * 获取原子状态
   */
  const readAtomState = (atom) => {
    const atomState = ensureAtomState(atom)
    const prevEpochNumber = atomState.n
    try {
      return atomState
    } catch (error) {
      delete atomState.v
      atomState.e = error
      ++atomState.n
      return atomState
    } finally {
      // 版本不一致，说明有更新或者错误
      if (prevEpochNumber !== atomState.n) {
        changedAtoms.add(atom)
      }
    }
  }

  /**
   * 设置原子值
   */
  const setAtomStateValueOrPromise = (atom, valueOrPromise, ensureAtomState) => {
    const atomState = ensureAtomState(atom)
    const hasPrevValue = 'v' in atomState
    const prevValue = atomState.v
    atomState.v = valueOrPromise
    delete atomState.e
    // 值发生变化，版本号+1
    if (!hasPrevValue || !Object.is(prevValue, atomState.v)) {
      ++atomState.n
    }
  }

  /**
   * 修改原子状态
   */
  const writeAtomState = (atom, ...args) => {
    const getter = (a) => returnAtomValue(readAtomState(a))
    const setter = (a, ...args) => {
      const aState = ensureAtomState(a)
      if (isSelfAtom(atom, a)) {
        const prevEpochNumber = aState.n
        const v = args[0]
        setAtomStateValueOrPromise(a, v, ensureAtomState) // 设置原子值
        mountDependencies(a) // 挂载依赖
        // 版本不一致，说明有更新或者错误
        if (prevEpochNumber !== aState.n) {
          changedAtoms.add(a)
        }
        return undefined
      } else {
        return writeAtomState(a, ...args)
      }
    }
    return atomWrite(atom, getter, setter, ...args)
  }

  const store = {
    get: (atom) => returnAtomValue(readAtomState(atom)),
    set: (atom, ...args) => {
      return writeAtomState(atom, ...args)

    sub: (atom, listener) => {
      const mounted = mountAtom(atom)
```

```JavaScript
  storeHooks.m.add(undefined, (atom) => {

    debugMountedAtoms.add(atom)
    const atomState = atomStateMap.get(atom)
    // 兼容 DevStoreRev4
    atomState.m = mountedAtoms.get(atom)
  })
  storeHooks.u.add(undefined, (atom) => {
    debugMountedAtoms.delete(atom)
    const atomState = atomStateMap.get(atom)
    // 兼容 DevStoreRev4
    delete atomState.m
  })

  const devStore = {
    // 高级调试，查看所有原子状态
    dev4_get_internal_weak_map: () => {
```
